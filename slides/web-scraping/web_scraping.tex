\section{Look first}
\begin{frame}[fragile]{Rule number one}
\begin{beamercolorbox}[rounded=true,sep=0.9em,wd=\linewidth]{block body alerted}
  \centering\large
  \textbf{Never write a scraper for a page you have not read.}
\end{beamercolorbox}

\vspace{0.9em}
Before any code, answer three questions by looking:
\begin{enumerate}
  \item \textbf{What exactly do I want?} Point at it on the screen.
  \item \textbf{Where does it sit} in the page?
  \item \textbf{How does the page get it?} Is it already in the page, or
        fetched separately?
\end{enumerate}

\vspace{0.6em}
\takeaway{Ten minutes of looking saves an afternoon of guessing.}
\end{frame}

\section{How the web works}
\begin{frame}[fragile]{Every scrape is the same two steps}
\begin{columns}[T,onlytextwidth]
  \column{0.46\textwidth}
  \begin{block}{1. Request}
    You ask a server for an address (URL). You may attach headers saying who
    you are.
  \end{block}
  \begin{block}{2. Reply}
    The server sends back a status code and some content.
  \end{block}

  \column{0.50\textwidth}
  \textbf{Status codes worth knowing}
  \vspace{0.3em}
  \begin{itemize}\small
    \item \statebadge{LightGreen}{200} fine, here it is
    \item \statebadge{LightOrange}{403} I know who you are, and no
    \item \statebadge{LightOrange}{404} no such thing
    \item \statebadge{LightRed}{429} you are asking too fast
    \item \statebadge{LightRed}{500} the server broke
  \end{itemize}

  \vspace{0.5em}
  \small Always check the code. A 404 page still ``downloads'' perfectly
  happily, and will silently produce nonsense.
\end{columns}

\vspace{0.6em}
\takeaway{Scraping is just doing by hand what your browser does automatically.}
\end{frame}

% =============================================================================
\begin{frame}[fragile]{Two kinds of reply}
\begin{columns}[T,onlytextwidth]
  \column{0.48\textwidth}
  \textbf{HTML} --- built for eyes
\begin{lstlisting}[style=code]
<div class="pay-input">
  <bdi>$300,000</bdi>
  <span>-</span>
  <bdi>$405,000</bdi>
</div>
\end{lstlisting}
  \small Layout and content mixed together. You must dig the values out, and
  the digging breaks when the design changes.

  \column{0.48\textwidth}
  \textbf{JSON} --- built for programs
\begin{lstlisting}[style=code]
{
  "min_cents": 30000000,
  "max_cents": 40500000,
  "currency_type": "USD"
}
\end{lstlisting}
  \small Names and values, nothing else. It becomes a Python dictionary in
  one line.
\end{columns}

\vspace{0.7em}
\takeaway{If a JSON version exists, use it. It is faster, cleaner, and far
more stable.}
\end{frame}

% =============================================================================
\begin{frame}[fragile]{The decision tree}
\begin{enumerate}
  \item \textbf{Is there a bulk download?} Take it and stop. (Statistical
        agencies, World Bank, Eurostat.)
  \item \textbf{Is there an official API?} Use it. Documented and stable.
  \item \textbf{Is there a hidden JSON address?} \statebadge{LightGreen}{we are here}
        Check the Network tab. Very often yes.
  \item \textbf{Is the content in the HTML?} Then \texttt{requests} +
        \texttt{BeautifulSoup}.
  \item \textbf{Does the page need a real browser?} Then Selenium. Slow, but
        it always works.
  \item \textbf{Is the site actively blocking you?} Reconsider the question,
        not the tooling.
\end{enumerate}

\vspace{0.7em}
\takeaway{Work down the list. Most people start at step 4 and never discover
that step 3 would have done.}
\end{frame}

\section{Three ways}
\begin{frame}[fragile]{Method 1: request the data directly}
\begin{columns}[T,onlytextwidth]
  \column{0.60\textwidth}
\begin{lstlisting}[style=py]
import requests

url = ("https://boards-api.greenhouse.io/v1"
       "/boards/anthropic/jobs/4981828008"
       "?pay_transparency=true")

r = requests.get(url, timeout=60)
r.raise_for_status()        # stop if not 200

job = r.json()              # -> a Python dict
pay = job["pay_input_ranges"][0]
\end{lstlisting}

  \column{0.38\textwidth}
  \textbf{Output}
\begin{lstlisting}[style=terminal]
job["title"]
  Senior+ Software
  Engineer, ...

pay["min_cents"]/100
  300000.0
pay["max_cents"]/100
  405000.0
\end{lstlisting}
\end{columns}

\vspace{0.2em}
\takeaway{No parsing: \texttt{.json()} returns a dictionary and we read
fields from it.}

\vspace{0.2em}
{\scriptsize\color{Muted} Inspect live:
\texttt{boards-api.greenhouse.io/v1/boards/anthropic/jobs/4981828008?pay\_transparency=true}}
\end{frame}

\begin{frame}[fragile]{Method 2: download the page, parse the HTML}
\begin{columns}[T,onlytextwidth]
  \column{0.60\textwidth}
\begin{lstlisting}[style=py]
import requests
from bs4 import BeautifulSoup

html = requests.get(url_web, timeout=60).text
soup = BeautifulSoup(html, "html.parser")

title = soup.find("h1").get_text(strip=True)

# each <bdi> tag holds a value
texts = [t.get_text(strip=True)
         for t in soup.find_all("bdi")]

amounts = [float(t.replace("$","")
           .replace(",",""))
           for t in texts if t.startswith("$")]
\end{lstlisting}

  \column{0.38\textwidth}
  \textbf{Output}
\begin{lstlisting}[style=terminal]
texts
  ['$300,000',
   '$405,000',
   'USD']

amounts
  [300000.0,
   405000.0]
\end{lstlisting}
\end{columns}

\vspace{0.2em}
\takeaway{Works, but relies on the salary sitting in a \texttt{<bdi>} tag. If
the page design changes, this can break without warning.}
\end{frame}

\begin{frame}[fragile]{Method 3: drive a real browser}
\begin{lstlisting}[style=py]
from selenium import webdriver
from selenium.webdriver.chrome.options import Options
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC

options = Options()
options.add_argument("--headless=new")     # run with no visible window
driver = webdriver.Chrome(options=options)

driver.get("https://job-boards.greenhouse.io/anthropic/jobs/4981828008")

WebDriverWait(driver, 20).until(            # <- do not skip this
    EC.presence_of_element_located((By.TAG_NAME, "bdi")))

texts = [e.text for e in driver.find_elements(By.TAG_NAME, "bdi")]
driver.quit()                               # always close it
\end{lstlisting}

\vspace{0.2em}
\takeaway{Selenium is the tool that always works, and the one to reach for
last.}
\end{frame}

% =============================================================================
\begin{frame}[fragile]{When you actually need Selenium}
\begin{columns}[T,onlytextwidth]
  \column{0.48\textwidth}
  \begin{block}{Use it when}
    \begin{itemize}\small
      \item the page is empty until JavaScript fills it
      \item you must click, scroll, or log in
      \item content appears only after interaction
    \end{itemize}
  \end{block}

  \column{0.48\textwidth}
  \begin{beamercolorbox}[rounded=true,sep=0.7em]{block body alerted}
    \textbf{The cost}
    \begin{itemize}\small
      \item a whole browser per run
      \item slow, and heavy on memory
      \item breaks in new ways (see next slide)
      \item needs a matching driver
    \end{itemize}
  \end{beamercolorbox}
\end{columns}

\vspace{0.7em}
\begin{beamercolorbox}[rounded=true,sep=0.6em,wd=\linewidth]{block body}
  \small \textbf{Good news:} Selenium 4 downloads the right driver for you.
  Older guides tell you to install \texttt{chromedriver} by hand---you do not
  have to, provided you are on \texttt{selenium>=4.25}.
\end{beamercolorbox}
\end{frame}

% =============================================================================
\begin{frame}[fragile]{The error you will meet in your first ten minutes}
\begin{lstlisting}[style=terminal]
selenium.common.exceptions.StaleElementReferenceException:
Message: stale element reference: stale element not found
\end{lstlisting}

\vspace{0.5em}
\begin{columns}[T,onlytextwidth]
  \column{0.47\textwidth}
  \textbf{What happened}\\
  \small You found an element. The page then rebuilt itself and threw that
  element away. Your variable now points at something that no longer exists.

  \column{0.49\textwidth}
  \textbf{The fix}\\
  \small Wait for the page to settle, then look the element up \emph{again}.
\begin{lstlisting}[style=py]
for attempt in range(3):
    try:
        title = driver.find_element(
            By.TAG_NAME, "h1").text
        break
    except StaleElementReferenceException:
        time.sleep(1)
\end{lstlisting}
\end{columns}

\vspace{0.5em}
\takeaway{This is not your bug. It is what browsers do. Write for it from the
start.}
\end{frame}

\section{Scaling up}
\begin{frame}[fragile]{From 3 jobs to 9{,}000: four new problems}
\begin{columns}[T,onlytextwidth]
  \column{0.48\textwidth}
  \begin{block}{1. Things fail}
    \small Boards close, names change, connections drop. One failure must not
    kill the run: catch the error, return empty, carry on.
  \end{block}
  \begin{block}{2. Waiting dominates}
    \small Your program is idle almost all the time. Run several requests at
    once with a thread pool.
  \end{block}

  \column{0.48\textwidth}
  \begin{block}{3. Re-running wastes everything}
    \small Save every raw reply to disk. The second run costs nothing and
    makes no new requests.
  \end{block}
  \begin{block}{4. You are a guest}
    \small Identify yourself. Keep the number of workers modest. Take the
    bulk endpoint when it exists.
  \end{block}
\end{columns}

\vspace{0.6em}

\section{What the data says}
\begin{frame}{Background: compensating differentials}
Compensating differential theory (Smith; Rosen, 1986): in equilibrium,
undesirable job attributes should be associated with higher pay, desirable
ones with lower pay.

\vspace{0.6em}
Estimating this from wage data is difficult. A common concern: more
productive workers tend to obtain both higher pay \emph{and} better working
conditions, and worker productivity is not fully observed. Estimated
differentials can therefore come out with an unexpected sign.

\vspace{0.6em}
\begin{beamercolorbox}[rounded=true,sep=0.7em]{block body}
  \small \textbf{Why postings are a useful complement.} A posted range is
  attached to a vacancy, not to a hired worker, so worker selection does not
  enter the observation in the same way.
\end{beamercolorbox}
\end{frame}

\section{The text is data too}
\begin{frame}{So far we used one number per posting}
A posting is mostly \emph{text}: what the job requires, what it offers, which
tools it uses. That text can be turned into variables.

\vspace{0.7em}
\begin{beamercolorbox}[rounded=true,sep=0.7em,wd=\linewidth]{block body}
  \textbf{Question.} Which technical skills are associated with higher posted
  pay, holding the employer and seniority fixed?
\end{beamercolorbox}

\vspace{0.7em}
The difficulty is not extracting words. It is doing so in a way you could
defend in a seminar---because if you choose the terms after seeing which ones
correlate with pay, the exercise is circular.
\end{frame}

% =============================================================================
\begin{frame}[fragile]{Three rules for turning text into variables}
\begin{enumerate}
  \item \textbf{Do not invent the list.} Take an external, published
        taxonomy. We use the \textbf{O*NET Hot Technologies} list
        (170 technologies), so the attribute set is not ours to choose.
  \item \textbf{Read the right section.} A tool named under
        \emph{Requirements} is a requirement; the same word in the company
        blurb is not. We split the posting on its headers.
  \item \textbf{Handle negation.} Consider this real sentence:
\begin{lstlisting}[style=terminal]
"No ML or Research experience is required"
\end{lstlisting}
        A naive keyword search records this as an ML requirement---with the
        sign exactly backwards.
\end{enumerate}

\vspace{0.4em}
\takeaway{Also beware short names that are ordinary words: matching
\texttt{access}, \texttt{cloud} or \texttt{team} tags most of the sample. For
those we require the full product name.}
\end{frame}

\section{Responsibly}
\begin{frame}[fragile]{Scraping responsibly}
\begin{columns}[T,onlytextwidth]
  \column{0.48\textwidth}
  \textbf{Practical}
  \begin{itemize}\small
    \item prefer a bulk download or an API
    \item send a \texttt{User-Agent} saying who you are
    \item ask for many records per request
    \item keep concurrency modest; back off on \statebadge{LightRed}{429}
    \item cache raw replies
  \end{itemize}

  \column{0.48\textwidth}
  \textbf{Legal and ethical}
  \begin{itemize}\small
    \item read \texttt{robots.txt} and the terms
    \item public $\neq$ free of obligations
    \item do not collect personal data you do not need
    \item GDPR applies to people, not to job adverts
    \item never scrape behind a login you agreed not to
  \end{itemize}
\end{columns}

\vspace{0.8em}
\takeaway{Keeping raw responses reduces load on the server and makes your
results reproducible.}
\end{frame}

\begin{frame}[fragile]{Appendix: politeness in code}
\begin{lstlisting}[style=py]
import time, requests

SESSION = requests.Session()                 # reuse the connection
SESSION.headers.update({"User-Agent": "CEMFI research (you@email.com)"})

def get(url, tries=4):
    for attempt in range(tries):
        r = SESSION.get(url, timeout=45)
        if r.status_code == 429:             # too fast: wait longer each time
            time.sleep(2 ** attempt)
            continue
        if r.status_code >= 500:             # server problem: try again
            time.sleep(1 + attempt)
            continue
        return r
    return None
\end{lstlisting}

\vspace{0.3em}
\takeaway{Exponential backoff: wait 1s, then 2s, then 4s. Hammering a
struggling server is how you get blocked.}
\end{frame}

\begin{frame}[fragile]{Appendix: a session is faster than repeated \texttt{get}}
\begin{columns}[T,onlytextwidth]
  \column{0.48\textwidth}
\begin{lstlisting}[style=py]
# new connection every time
for url in urls:
    requests.get(url)
\end{lstlisting}
  \small Each call re-negotiates encryption with the server. On this site that
  roughly \textbf{doubled} the time per page.

  \column{0.48\textwidth}
\begin{lstlisting}[style=py]
# one connection, reused
s = requests.Session()
for url in urls:
    s.get(url)
\end{lstlisting}
  \small Also remembers headers and cookies, so you set them once.
\end{columns}

\vspace{0.8em}
\begin{beamercolorbox}[rounded=true,sep=0.6em,wd=\linewidth]{block body}
  \small We hit this while preparing the session: BeautifulSoup looked
  \emph{slower than Selenium} until we used a \texttt{Session}. Selenium was
  reusing its connection; our \texttt{requests} code was not.
\end{beamercolorbox}
\end{frame}
