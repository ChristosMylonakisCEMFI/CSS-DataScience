\begin{frame}{Prediction as an objective}
\begin{wideitemize}
  \item A rule: observed characteristics in, an unobserved outcome out.
  \item Estimated on one sample, judged on another.
  \item The criterion is error on the second, not the value of any coefficient.
  \item Bias in the coefficients is acceptable when it lowers that error.
\end{wideitemize}
\closingline{Everything that follows is about estimating that error honestly.}
\end{frame}

\section{Overfitting and the trade-off}

\begin{frame}{Overfitting}
\simfig{0.68\textwidth}{fig_overfit_reg}
\begin{wideitemize}
  \item In-sample fit: better at every degree.
  \item 20 observations, degree 19: every point exactly.
  \item Error in an independent sample: lowest at a low degree.
\end{wideitemize}
\closingline{Fit in the estimation sample is not evidence of accuracy elsewhere.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Overfitting with a binary outcome}
\simfig{0.78\textwidth}{fig_overfit_clf}
\begin{wideitemize}
  \item Overfitting is not a property of polynomials or of continuous outcomes.
  \item Left unrestricted, the tree keeps splitting until each region is pure.
  \item It ends up isolating single observations, and misclassifies new ones.
\end{wideitemize}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Bias and variance}
\simfig{0.74\textwidth}{fig_bias_variance}
\begin{wideitemize}
  \item Each grey line: one independent sample of 25 observations.
  \item Restrictive fit: stable across samples, wrong on average.
  \item Flexible fit: right on average, far off in any one sample.
\end{wideitemize}
\closingline{Two distinct errors, and less of one buys more of the other.}
\end{frame}

\section{Estimating out-of-sample error}

\begin{frame}{Holding observations back}
\begin{center}
\begin{tikzpicture}[font=\small]
  \draw[fill=white,draw=black] (0,0) rectangle (6.0,0.8);
  \node at (3.0,0.4) {Training};
  \draw[fill=PaleGray,draw=black] (6.0,0) rectangle (8.6,0.8);
  \node at (7.3,0.4) {Validation};
  \draw[fill=FigGray,draw=black] (8.6,0) rectangle (11.2,0.8);
  \node[white] at (9.9,0.4) {Test};
\end{tikzpicture}
\end{center}
\vspace{0.3em}
\begin{wideitemize}
  \item Out-of-sample error is unobserved. Hide observations and imitate it.
  \item Training: estimate the coefficients of one given specification.
  \item Validation: measure error there to \emph{choose} --- which regressors,
        which estimator, how flexible.
  \item Test: measure error once more, to \emph{report} what the chosen rule
        will do.
\end{wideitemize}
\closingline{Both measure error; they answer different questions.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Why the test sample is separate}
\begin{wideitemize}
  \item Compare twenty specifications on the validation sample. One wins.
  \item It won partly on merit, partly because that sample's noise suited it.
  \item So the winner's validation error is better than its true accuracy.
  \item The more specifications compared, the larger that gap.
\end{wideitemize}
\closingline{The test sample took part in no comparison, so its error is not
flattered by having been chosen on. That is the number to report.}
\end{frame}

\begin{frame}{Cross-validation}
\begin{center}
\begin{tikzpicture}[font=\scriptsize,y=0.50cm]
  \foreach \r in {0,1,2,3,4} {
    \foreach \c in {0,1,2,3,4} {
      \pgfmathparse{\c==\r ? "FigRed" : "FigBlue"}\edef\col{\pgfmathresult}
      \draw[fill=\col,draw=black,line width=0.3pt]
        (\c*1.0,-\r) rectangle (\c*1.0+0.9,-\r+0.46);
    }
    \node[anchor=west] at (5*1.0+0.1,-\r+0.23) {fold \the\numexpr\r+1\relax};
  }
  \draw[fill=FigRed,draw=black,line width=0.3pt] (7.4,-0.9) rectangle (7.8,-0.5);
  \node[anchor=west] at (7.85,-0.7) {validation};
  \draw[fill=FigBlue,draw=black,line width=0.3pt] (7.4,-2.1) rectangle (7.8,-1.7);
  \node[anchor=west] at (7.85,-1.9) {estimation};
\end{tikzpicture}
\end{center}
\vspace{0.2em}
\begin{wideitemize}
  \item One validation block wastes those observations, and is one arbitrary draw.
  \item Instead, cut the training data into $k$ folds of equal size.
  \item Each fold takes the validation role in turn; the other $k-1$ estimate.
  \item Average the $k$ errors: one number per specification, using all the data.
\end{wideitemize}
\closingline{The validation role rotates inside the training data; nothing is
permanently set aside. The test sample stays outside the rotation, still unused.}
\end{frame}

\section{Leakage}

\begin{frame}{What leakage is}
\begin{wideitemize}
  \item Leakage: the fit uses information that will not be available when the
        rule is actually applied.
  \item The estimate of accuracy is then an estimate of a task that is easier
        than the real one.
  \item It shows up as a good cross-validated error and a disappointing one in use.
\end{wideitemize}
\closingline{Nothing in the procedure complains. The fold was held out, so the
arithmetic is correct; it is the data that were contaminated.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Leakage in a cross-validated fit}
\begin{columns}[T,onlytextwidth]
  \column{0.44\textwidth}
  \vspace{0.6em}
  \includegraphics[width=\linewidth]{fig_leakage}

  \column{0.52\textwidth}
  \vspace{0.6em}
  \begin{wideitemize}
    \item Here one regressor was built from the outcome itself.
    \item Cross-validation reports an almost perfect fit.
    \item Remove that regressor and the fit is ordinary.
  \end{wideitemize}
\end{columns}
\closingline{The held-out fold was held out, but its outcome had already
travelled into the regressors.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Two kinds of leakage}
\begin{wideitemize}
  \item Wrong date: a regressor recorded after the outcome, or unknown at the
        moment the prediction would be made.
  \item Wrong sample: a quantity computed on the whole data and then used
        inside the folds.
  \item The second is the common one: standardization, imputation, category
        means, the choice of which regressors to keep.
\end{wideitemize}
\closingline{Both make the cross-validated error too low, and the error in use
a surprise.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Keeping the estimate honest}
\begin{wideitemize}
  \item Split first, transform afterwards. Never the reverse.
  \item Means, standard deviations and imputed values are estimated on the
        training fold, then applied to the held-out one.
  \item A regressor enters only if it is known before the outcome is realized.
  \item The test sample is opened once, at the end.
\end{wideitemize}
\closingline{In code, this is why the transformations and the estimator are
fitted together, inside the fold, rather than beforehand.}
\end{frame}

\section{Measures of accuracy}

\begin{frame}{Measuring accuracy: continuous outcomes}
\simfig{0.78\textwidth}{fig_rmse_mae}
\begin{wideitemize}
  \item Root mean squared error: large errors count for more.
  \item Mean absolute error: every unit of error counts the same.
  \item A few extreme observations move the first, not the second.
\end{wideitemize}
\closingline{Fix the measure before the estimation, by the cost of an error.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Measuring accuracy: binary outcomes}
\simfig{0.76\textwidth}{fig_accuracy_trap}
\begin{wideitemize}
  \item Rare ones: the share of correct classifications says little.
  \item Zero everywhere: 96\% correct, nothing identified.
  \item Predicted against actual, as a table: the useful summary.
\end{wideitemize}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Thresholds, precision, and recall}
\begin{wideitemize}
  \item Estimators return a probability. A classification needs a threshold.
  \item Precision: of those predicted one, the share that are one.
  \item Recall: of those that are one, the share predicted one.
  \item A higher threshold: more precision, less recall.
\end{wideitemize}
\closingline{The threshold is a decision about the cost of the two errors,
not an estimation problem.}
\end{frame}

\end{frame}

\section{Constructing regressors}

\begin{frame}{An estimator adds or splits; it never combines}
\simfig{0.80\textwidth}{fig_combine}
\begin{wideitemize}
  \item A linear model is additive; a tree is a step function on rectangles.
  \item Neither forms a ratio, a group mean, or a rank within a group.
  \item Both hold the ingredients here, and fifteen irrelevant regressors besides.
  \item Handing over the combination is what moves each of them.
\end{wideitemize}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Which combinations}
\begin{wideitemize}
  \item Ratios: debt to assets, price to earnings, output per worker.
  \item Deflated, real and per-capita terms, so units compare across time and size.
  \item Differences and growth rates: the level and the change are different
        regressors.
  \item Position within a group: size against the industry median, income
        against the region, a rank within the cohort.
\end{wideitemize}
\closingline{An economist can write these down in a morning. No estimator will
discover them in any amount of data.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Three standard constructions}
\begin{wideitemize}
  \item Principal components: many correlated regressors replaced by a few.
        Chosen without reference to the outcome, so the first need not predict.
  \item Lags, differences and rolling means, where the data have a time
        dimension --- each dated to exist at the moment of the prediction.
  \item The outcome itself: fitting to $\log y$ penalizes relative error rather
        than absolute.
\end{wideitemize}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Did the regressor earn its place?}
\begin{wideitemize}
  \item Shuffle one regressor, leave the rest alone, and see how much the
        cross-validated error worsens.
  \item Nothing worsens: the regressor carries nothing, whatever its coefficient.
  \item Almost everything worsens under one regressor: the signature of leakage.
  \item Any estimator, and it reports on the regressor rather than on a coefficient.
\end{wideitemize}
\closingline{This is what the constructions of the last three slides are to be
put through.}
\end{frame}

\section{Choosing an estimator}

\begin{frame}{Three decisions}
\begin{wideitemize}
  \item Which family: penalized linear regression, or something tree-based.
  \item How flexible within that family: the penalty, the depth, the number of trees.
  \item When to stop: a gain smaller than the variation across folds is not a gain.
\end{wideitemize}
\closingline{All three are settled by the same number, the cross-validated
error, computed on identical folds with identical regressors. Least squares
is the first entry, as the benchmark the rest have to beat.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Restrictive fits and flexible fits}
\vspace{0.6em}
\begin{center}

\textbf{Restrictive fits win} & \textbf{Flexible fits win} \\

\end{center}
\closingline{Most outcomes in economics sit on the left: many regressors,
modest signal. Each row is a conjecture, and the folds settle it.}
\end{frame}

\begin{frame}{Penalized regression}
\simfig{0.56\textwidth}{fig_lasso_path}
\begin{wideitemize}
  \item 70 observations, 40 regressors, 3 with nonzero coefficients.
  \item The penalty shrinks the coefficients, and sets the irrelevant to zero.
  \item Lowest test error at a strictly positive penalty.
  \item Which regressors survive is not a finding: another sample selects others.
\end{wideitemize}
\closingline{Penalty, depth, number of trees: each is a choice, and the folds make it.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{The random forest as a default}
\simfig{0.60\textwidth}{fig_tree_rf}
\begin{wideitemize}
  \item One tree: the boundary is a feature of the particular sample.
  \item A forest averages many trees on resampled data. Most of the variance goes.
  \item Two choices, and little turns on either. Nothing to scale or transform.
  \item One estimator for a continuous outcome and for a binary one.
\end{wideitemize}
\source{Fern\'andez-Delgado et al.\ (2014) compared 179 classifiers from 17
families over 121 data sets. The random forests came out best.}
\end{frame}

\simfig{0.66\textwidth}{fig_compare}
\begin{wideitemize}
  \item Same regressors, same folds, same measure. Bars: the average over five
        folds; whiskers: one standard error.
  \item Neither outcome is generated by a tree: on the left $2\sin x_1 + x_2^2
        - 1.5x_3 + x_1x_4$, on the right two interleaved crescents.
  \item The tree-based fits win anyway. On data close to linear the ordering
        reverses.
\end{wideitemize}
\closingline{Read off the whiskers: boosting is ahead on the left; on the right
the two are within one standard error, and the forest is simpler.}

\begin{frame}{Summary}
\begin{wideitemize}
  \item Accuracy concerns observations not used in the estimation.
  \item In-sample fit does not measure it; bias against variance is the trade-off.
  \item Cross-validation estimates it. Leakage quietly destroys the estimate.
  \item Fix the measure of accuracy in advance, by the cost of an error.
  \item Regressors first, estimator second.
\end{wideitemize}
\end{frame}

\begin{frame}{The decomposition of expected prediction error}
For a new observation at $x_0$, with $y = f(x_0) + \varepsilon$ and $\hat f$ estimated on a
random sample,
\[
  \mathbb{E}\!\left[(y - \hat f(x_0))^2\right]
  = \big(\mathbb{E}[\hat f(x_0)] - f(x_0)\big)^2
  + \operatorname{Var}\!\big(\hat f(x_0)\big)
  + \operatorname{Var}(\varepsilon).
\]
\begin{wideitemize}
  \item The expectation is taken over samples, not over observations.
  \item The first two terms can be traded against one another.
  \item The third cannot be reduced by any estimator.
  \item Penalization and averaging lower the second at some cost in the first.
\end{wideitemize}
\end{frame}

\begin{frame}{Boosting: successive fits}
\simfig{0.80\textwidth}{fig_boosting}
\begin{wideitemize}
  \item A single tree of depth two is a coarse step function.
  \item Successive trees fitted to the residuals accumulate into an accurate fit.
  \item Continued far enough, the procedure begins to fit the noise.
  \item The number of trees is itself chosen by cross-validation.
\end{wideitemize}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Bet on sparsity}
\begin{wideitemize}
  \item If many regressors each matter a little, no procedure predicts well:
        the data cannot support that many estimated effects.
  \item If a few matter a lot, penalized methods find them.
  \item So act as though few matter. Build the candidates, let the penalty select.
  \item The bet is not always right. It is that there is nothing better to bet on.
\end{wideitemize}
\source{Hastie, Tibshirani and Friedman, sec.~16.2.2. Stated there as a
principle, not a theorem.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Transformations of a single regressor}
\simfig{0.50\textwidth}{fig_monotone}
\begin{wideitemize}
  \item A tree splits on $x \le t$, so it reads the ordering of $x$ and nothing else.
  \item An increasing transformation leaves its partition, and its fit, unchanged.
  \item The forest scores 0.30 either way; the same $\log$ is worth 0.15 to
        least squares.
\end{wideitemize}
\closingline{Rescaling one regressor: everything to a linear model, exactly
nothing to a tree. Combining two regressors: worth doing for both.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Categorical variables: the encoding is a decision}
\begin{wideitemize}
  \item Few levels: one indicator each, and nothing further is needed.
  \item Many levels: as many regressors as levels, most of them nearly empty,
        and each estimated on a handful of observations.
  \item Group the rare levels together, or replace a level by its frequency.
        The frequency uses no outcome, so it can be computed before the split.
  \item Replacing a level by its mean outcome predicts well and leaks: the
        outcome enters the regressor. Compute it within the fold or not at all.
\end{wideitemize}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Missing values carry information}
\begin{wideitemize}
  \item Values are rarely missing at random: no credit history, a firm too
        young to report, a question refused.
  \item Impute the value, and add an indicator for having imputed it.
  \item The indicator is often the better predictor of the two.
  \item Both the imputation and the indicator are built inside the fold.
\end{wideitemize}
\closingline{Dropping the incomplete observations throws that information away,
and leaves a sample selected on the regressors.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Further measures for binary outcomes}
\begin{wideitemize}
  \item When ones are rare, the precision--recall curve is more informative.
  \item ROC can look favourable when few of the predicted ones are correct.
  \item The $F_1$ measure combines precision and recall in one number.
  \item Log-loss penalizes confident predictions that turn out to be wrong.
  \item Calibration: of those given probability $0.7$, about 70\% should be ones.
\end{wideitemize}
\end{frame}
