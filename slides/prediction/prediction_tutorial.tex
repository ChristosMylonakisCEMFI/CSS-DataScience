\begin{frame}{Prediction as an objective}
\begin{wideitemize}
  \item A rule: observed characteristics in, an unobserved outcome out.
  \item Estimated on one sample, judged on another.
  \item The criterion is error on the second, not the value of any coefficient.
  \item Bias in the coefficients is acceptable when it lowers that error.
\end{wideitemize}
\closingline{Everything that follows is about estimating that error honestly.}
\end{frame}

\begin{frame}{Overfitting}
\simfig{0.68\textwidth}{fig_overfit_reg}
\begin{wideitemize}
  \item In-sample fit: better at every degree.
  \item 20 observations, degree 19: every point exactly.
  \item Error in an independent sample: lowest at a low degree.
\end{wideitemize}
\closingline{Fit in the estimation sample is not evidence of accuracy elsewhere.}
\end{frame}

\end{frame}

\simfig{0.74\textwidth}{fig_bias_variance}

  \item Restrictive fit: stable across samples, wrong on average.
  \item Flexible fit: right on average, far off in any one sample.
\end{wideitemize}
\closingline{Two distinct errors, and less of one buys more of the other.}

\section{Estimating out-of-sample error}

\begin{frame}{Holding observations back}
\begin{center}
\begin{tikzpicture}[font=\small]
  \draw[fill=white,draw=black] (0,0) rectangle (6.0,0.8);
  \node at (3.0,0.4) {Training};
  \draw[fill=PaleGray,draw=black] (6.0,0) rectangle (8.6,0.8);
  \node at (7.3,0.4) {Validation};
  \draw[fill=FigGray,draw=black] (8.6,0) rectangle (11.2,0.8);
  \node[white] at (9.9,0.4) {Test};
\end{tikzpicture}
\end{center}
\vspace{0.3em}
\begin{wideitemize}
  \item Out-of-sample error is unobserved. Hide observations and imitate it.
  \item Training: the coefficients, for one given specification.
  \item Validation: the choice between specifications, and how flexible.
  \item Test: one number, once, after every choice is final.
\end{wideitemize}
\closingline{Why a third sample: a sample used to choose is no longer unseen,
and its error is optimistic.}
\end{frame}

  }
  \draw[fill=FigRed,draw=black,line width=0.3pt] (7.4,-0.9) rectangle (7.8,-0.5);
  \node[anchor=west] at (7.85,-0.7) {validation};
  \draw[fill=FigBlue,draw=black,line width=0.3pt] (7.4,-2.1) rectangle (7.8,-1.7);
  \node[anchor=west] at (7.85,-1.9) {estimation};
\end{tikzpicture}
\end{center}
\vspace{0.2em}
\begin{wideitemize}
  \item One validation sample: few observations, and one arbitrary draw.
  \item Instead, cut the training data into $k$ folds of equal size.
  \item Each fold is the validation sample once; the other $k-1$ estimate.
  \item Average the $k$ errors.
\end{wideitemize}
\closingline{Cross-validation replaces the validation sample, not the test
sample. That one is still untouched.}
\end{frame}

\section{Leakage}

\begin{frame}{Leakage}
\begin{columns}[T,onlytextwidth]
  \column{0.44\textwidth}
  \vspace{0.6em}
  \includegraphics[width=\linewidth]{fig_leakage}

  \column{0.52\textwidth}
  \vspace{0.6em}
  \begin{wideitemize}
    \item One regressor built from the outcome.
    \item Cross-validated fit: almost perfect.
    \item Drop it: no better than honest.
  \end{wideitemize}
\end{columns}
\closingline{The held-out fold was never really unseen, and nothing in the
procedure said so.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Two kinds of leakage}
\begin{wideitemize}
  \item Wrong date: a regressor unknown at the moment of the prediction.
  \item Wrong sample: a quantity computed on data outside the training fold.
  \item The second, in practice: standardization, imputation, category means.
  \item Both raise the cross-validated fit and lower the real one.
\end{wideitemize}
\closingline{An unexpectedly good fit is a symptom before it is a result.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Keeping the estimate honest}
\begin{wideitemize}
  \item Split first, transform afterwards. Never the reverse.
  \item Every transformation fitted on the training fold, applied to the other.
  \item A regressor enters only if it is known before the outcome.
  \item The test sample is opened once, at the end.
\end{wideitemize}
\closingline{In code: the transformations go inside the fold, with the estimator.}
\end{frame}

\section{Measures of accuracy}

\begin{frame}{Measuring accuracy: continuous outcomes}
\simfig{0.78\textwidth}{fig_rmse_mae}
\begin{wideitemize}
  \item Root mean squared error: large errors count for more.
  \item Mean absolute error: every unit of error counts the same.
  \item A few extreme observations move the first, not the second.
\end{wideitemize}
\closingline{Fix the measure before the estimation, by the cost of an error.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Measuring accuracy: binary outcomes}
\simfig{0.76\textwidth}{fig_accuracy_trap}
\begin{wideitemize}
  \item Rare ones: the share of correct classifications says little.
  \item Zero everywhere: 96\% correct, nothing identified.
  \item Predicted against actual, as a table: the useful summary.
\end{wideitemize}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Thresholds, precision, and recall}
\begin{wideitemize}
  \item Estimators return a probability. A classification needs a threshold.
  \item Precision: of those predicted one, the share that are one.
  \item Recall: of those that are one, the share predicted one.
  \item A higher threshold: more precision, less recall.
\end{wideitemize}
\closingline{The threshold is a decision about the cost of the two errors,
not an estimation problem.}
\end{frame}

\section{Constructing regressors}

\begin{frame}{The choice of regressors}
\simfig{0.70\textwidth}{fig_features}
\begin{wideitemize}
  \item Least squares on one regressor, in both panels.
  \item The outcome: $y = 2\log x + \varepsilon$.
  \item $x$ replaced by $\log x$: out-of-sample error down by an order of magnitude.
\end{wideitemize}
\closingline{Same estimator, same data. The transformation supplies the missing knowledge.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{What a transformation buys}
\begin{wideitemize}
  \item Linear models: curvature and interactions only where supplied.
  \item Trees: both on their own, but learned, and only with enough data.
  \item A monotone transformation: everything for the first, nothing for a tree.
  \item Ratios, shares, per capita, real terms: worth supplying to either.
\end{wideitemize}
\closingline{The candidates come from the subject. The folds say which ones earn a place.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Categories, dates, missing values}
\begin{wideitemize}
  \item Few categories: one indicator each. Many: group the rare ones.
  \item The mean outcome within a category: leakage, unless computed per fold.
  \item Dates: year, month, day of the week, as separate regressors.
  \item Missing: impute, and keep the indicator. Missingness is informative.
\end{wideitemize}
\closingline{A regressor with no content is not free: it is paid for in variance.}
\end{frame}

\section{Choosing an estimator}

\begin{frame}{Choosing an estimator}
\begin{wideitemize}
  \item No estimator dominates. The ranking is a property of the data.
  \item Least squares first: the benchmark against which flexibility is judged.
  \item Then one flexible estimator, on the same regressors and the same folds.
  \item More only when the gain is larger than the variation across folds.
\end{wideitemize}
\closingline{Between two within sampling error of each other, take the simpler one.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{What decides the ranking}
\vspace{0.6em}
\begin{center}
\begin{tabular}{@{}p{0.40\textwidth}p{0.40\textwidth}@{}}
\toprule
\textbf{Restrictive fits win} & \textbf{Flexible fits win} \\
\midrule
Few observations per regressor & Many observations \\[0.45em]
Weak signal, much noise & Strong signal \\[0.45em]
Smooth, additive relations & Thresholds and interactions \\[0.45em]
Extrapolation is required & Prediction inside the data \\
\bottomrule
\end{tabular}
\end{center}
\closingline{Each row is a conjecture about the data. The cross-validated error
settles it.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{What each estimator buys}
\begin{wideitemize}
  \item Least squares: a benchmark, with nothing to choose.
  \item Ridge and LASSO: a penalty on the coefficients; LASSO also selects.
  \item Regression tree: thresholds and interactions; on its own, unstable.
  \item Random forest: the tree's flexibility, most of its variance removed.
  \item Boosting: usually the most accurate, and the most to choose.
\end{wideitemize}
\closingline{Each has its counterpart for a binary outcome.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Penalized regression}
\simfig{0.73\textwidth}{fig_lasso_path}
\begin{wideitemize}
  \item 70 observations, 40 regressors, 3 with nonzero coefficients.
  \item The penalty shrinks the coefficients, and sets the irrelevant to zero.
  \item Lowest test error at a strictly positive penalty.
\end{wideitemize}
\closingline{The penalty is one more choice, and the folds make it.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Trees and random forests}
\simfig{0.74\textwidth}{fig_tree_rf}
\begin{wideitemize}
  \item One tree: the boundary is a feature of the particular sample.
  \item A forest: many trees on resampled data, averaged.
  \item The flexibility kept, most of the variance gone.
\end{wideitemize}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Boosting}
\begin{wideitemize}
  \item Trees in sequence, not in parallel.
  \item Each one fitted to the residuals of the current model.
  \item Each entering with a small weight, so progress is gradual.
  \item More parameters than a forest, and it overfits when they are careless.
\end{wideitemize}
\closingline{A forest asks fewer questions of you, and is the better place to start.}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Comparing estimators}
\simfig{0.78\textwidth}{fig_compare}
\begin{wideitemize}
  \item Same regressors, same folds, for every estimator.
  \item Flexible estimators win here: curvature and interaction in the data.
  \item On data close to linear, the ordering reverses.
\end{wideitemize}
\closingline{The table earlier is a guide to where to look. This figure is the answer.}
\end{frame}

\begin{frame}{Summary}
\begin{wideitemize}
  \item Accuracy concerns observations not used in the estimation.
  \item In-sample fit does not measure it; bias against variance is the trade-off.
  \item Cross-validation estimates it. Leakage quietly destroys the estimate.
  \item Fix the measure of accuracy in advance, by the cost of an error.
  \item Regressors first, estimator second.
\end{wideitemize}
\end{frame}

\begin{frame}{The decomposition of expected prediction error}
For a new observation at $x_0$, with $y = f(x_0) + \varepsilon$ and $\hat f$ estimated on a
random sample,
\[
  \mathbb{E}\!\left[(y - \hat f(x_0))^2\right]
  = \big(\mathbb{E}[\hat f(x_0)] - f(x_0)\big)^2
  + \operatorname{Var}\!\big(\hat f(x_0)\big)
  + \operatorname{Var}(\varepsilon).
\]
\begin{wideitemize}
  \item The expectation is taken over samples, not over observations.
  \item The first two terms can be traded against one another.
  \item The third cannot be reduced by any estimator.
  \item Penalization and averaging lower the second at some cost in the first.
\end{wideitemize}
\end{frame}

\begin{frame}{Boosting: successive fits}
\simfig{0.80\textwidth}{fig_boosting}
\begin{wideitemize}
  \item A single tree of depth two is a coarse step function.
  \item Successive trees fitted to the residuals accumulate into an accurate fit.
  \item Continued far enough, the procedure begins to fit the noise.
  \item The number of trees is itself chosen by cross-validation.
\end{wideitemize}
\end{frame}

\begin{frame}{Further measures for binary outcomes}
\begin{wideitemize}
  \item When ones are rare, the precision--recall curve is more informative.
  \item ROC can look favourable when few of the predicted ones are correct.
  \item The $F_1$ measure combines precision and recall in one number.
  \item Log-loss penalizes confident predictions that turn out to be wrong.
  \item Calibration: of those given probability $0.7$, about 70\% should be ones.
\end{wideitemize}
\end{frame}
